Backtracking Template from [labuladong](https://labuladong.gitbook.io/algo-en/iii.-algorithmic-thinking/subset_permutation_combination)

```py
def backtrack(nums, start, track):
    # End condition
    if (end condition):
        result.append(track[:]) # copy, or every result aliases the same list
        return
    for i in range(start, len(nums)):
        track.append(nums[i])
        backtrack(nums, i + 1, track)
        track.pop()
```

Time Complexity: `O(N!)`.

Space Complexity: `O(n)`.

:::tip
For more details on time and space complexity, read this post: https://stackoverflow.com/questions/65526515/how-to-identify-time-and-space-complexity-of-recursive-backtracking-algorithms-w

*Figure REC-1: Combinations for n = 4, k = 2. Each branch only picks numbers larger than the last one, so `[2, 1]` is never built. Amber nodes are results: the path reached length k.*

```py
def combine(self, n, k):
    res = []
    self.backtrack(n, k, 1, [], res)
    return res

def backtrack(self, n, k, start, track, res):
    if len(track) == k:
        res.append(track[:]) # copy: track keeps changing after this
        return
    for i in range(start, n+1): # Order matters, increasing
        track.append(i)
        self.backtrack(n, k, i + 1, track, res)
        track.pop()
```

*Figure REC-2: Permutations of [1, 2, 3]. Each branch picks any number not already on the path, so order matters and every leaf is distinct. Amber nodes are results.*

```py
def permute(self, nums):
    res = []
    self.backtrack(nums, [], res)
    return res

def backtrack(self, nums, track, res):
    if len(track) == len(nums):
        res.append(track[:]) # copy: track keeps changing after this
        return
    for i in nums: # Order does not matter so we loop through entire array
        if i in track: # Exclude numbers already in track
            continue
        track.append(i)
        self.backtrack(nums, track, res)
        track.pop()
```

### [Subsets II](https://leetcode.com/problems/subsets-ii/)

#### Solution: Backtracking

```py
def subsetsWithDup(self, nums):
    res = []
    nums.sort() # Sort so duplicates sit next to each other
    self.backtrack(nums, 0, [], res)
    return res

def backtrack(self, nums, start, track, res):
    res.append(track[:])
    for i in range(start, len(nums)):
        if i > start and nums[i] == nums[i-1]: # IMPT: Skips over duplicate subsets
            continue
        track.append(nums[i])
        self.backtrack(nums, i + 1, track, res)
        track.pop()
```
